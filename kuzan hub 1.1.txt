local function cleanupOldGuis()
    local containers = {}
    pcall(function()
        if gethui then table.insert(containers, gethui()) end
    end)
    pcall(function()
        if CoreGui then table.insert(containers, CoreGui) end
    end)

    for _, container in ipairs(containers) do
        for _, name in ipairs({"NullUI", "SkidHubToggleGui", "KuzanToggleGui", "KuzanCommunity_SkidUI"}) do
            pcall(function()
                local old = container:FindFirstChild(name)
                if old then old:Destroy() end
            end)
        end
    end
end
cleanupOldGuis()

local function getOrDownloadAsset(fileName: string, url: string, fallbackAsset: string)
    if isfile and isfile(fileName) and getcustomasset then
        local ok, asset = pcall(function() return getcustomasset(fileName) end)
        if ok and asset then return asset end
    end

    if url and url ~= "" and writefile and getcustomasset then
        local ok, data = pcall(function() return game:HttpGet(url) end)
        if ok and type(data) == "string" and #data > 500 then
            pcall(function() writefile(fileName, data) end)
            local assetOk, asset = pcall(function() return getcustomasset(fileName) end)
            if assetOk and asset then return asset end
        end
    end

    return fallbackAsset or ""
end

local function decodeBase64Bytes(data: string): string
    local nativeFns = {
        crypt and crypt.base64decode,
        crypt and crypt.base64 and crypt.base64.decode,
        syn and syn.crypt and syn.crypt.base64 and syn.crypt.base64.decode,
        base64_decode,
        base64decode
    }
    local clean = data:gsub("[^A-Za-z0-9+/=]", "")
    for _, fn in ipairs(nativeFns) do
        if type(fn) == "function" then
            local ok, res = pcall(fn, clean)
            if ok and type(res) == "string" and #res > 10000 then
                return res
            end
        end
    end

    local chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    local lut = {}
    for i = 1, #chars do
        lut[string.byte(chars, i)] = i - 1
    end
    local out = {}
    local idx = 1
    local n = #clean
    for i = 1, n, 4 do
        local b1 = lut[string.byte(clean, i)] or 0
        local b2 = lut[string.byte(clean, i + 1)] or 0
        local c3 = string.byte(clean, i + 2)
        local c4 = string.byte(clean, i + 3)
        local b3 = (c3 and c3 ~= 61) and (lut[c3] or 0) or 0
        local b4 = (c4 and c4 ~= 61) and (lut[c4] or 0) or 0

        local triple = b1 * 262144 + b2 * 4096 + b3 * 64 + b4
        local o1 = math.floor(triple / 65536) % 256
        local o2 = math.floor(triple / 256) % 256
        local o3 = triple % 256

        if c3 == 61 then
            out[idx] = string.char(o1)
        elseif c4 == 61 then
            out[idx] = string.char(o1, o2)
        else
            out[idx] = string.char(o1, o2, o3)
        end
        idx = idx + 1
    end
    return table.concat(out)
end

local function getEmbeddedWatermarkAsset(fallbackAsset: string): string
    local customAssetFn = getcustomasset or getsynasset
    local wmFile = "kuzan_watermark_v11.jpg"
    if customAssetFn and isfile and isfile(wmFile) then
        local validCache = true
        if readfile then
            local okR, raw = pcall(readfile, wmFile)
            validCache = (okR and type(raw) == "string" and #raw > 15000)
        end
        if validCache then
            local okA, asset = pcall(customAssetFn, wmFile)
            if okA and type(asset) == "string" and #asset > 0 then
                return asset
            end
        end
    end

    if customAssetFn and writefile then
        local okDec, imgBytes = pcall(decodeBase64Bytes, KUZAN_WATERMARK_B64)
        if okDec and type(imgBytes) == "string" and #imgBytes > 15000 then
            pcall(writefile, wmFile, imgBytes)
            local okA, asset = pcall(customAssetFn, wmFile)
            if okA and type(asset) == "string" and #asset > 0 then
                return asset
            end
        end
    end

    return fallbackAsset or ""
end

local function executeScript(code: string, scriptName: string)
    local loader = loadstring or (load_string and load_string)
    if not loader then
        notify("Lỗi", "Executor không hỗ trợ loadstring!", 3.5)
        return
    end

    if autoCloseUI then
        hideKuzanHub()
    end

    notify("Đang chạy", "Đang khởi chạy " .. scriptName .. "... (Chạm Logo hoặc nhấn Ctrl để mở lại)", 3.5)
    task.spawn(function()
        local success, err = pcall(function()
            local fn = loader(code)
            if fn then fn() end
        end)
        if success then
            notify("Thành công", scriptName .. " đã bật thành công!", 3)
        else
            notify("Lỗi", tostring(err):sub(1, 55) .. "...", 4)
        end
    end)
end

local function buildGameTab(tabName, iconAsset, isCustom, gameData)
    local tab = rawWin:AddTab({ Name = tabName, Icon = iconAsset })
    tab._isCustomIcon = (isCustom == true)
    gameData = gameData or {}

    local defaultSections = {
        { Key = "Script Main", EmptyText = "Hiện chưa có script Main cho mục này." },
        { Key = "Script Auto", EmptyText = "Hiện chưa có script Auto (Sắp cập nhật)." },
        { Key = "Script Kaitun", EmptyText = "Hiện chưa có script Kaitun (Sắp cập nhật)." }
    }

    local handledKeys = {}
    for _, secInfo in ipairs(defaultSections) do
        handledKeys[secInfo.Key] = true
        local sec = createCollapsibleSection(tab, secInfo.Key, false)
        local list = gameData[secInfo.Key]
        if list and #list > 0 then
            for _, item in ipairs(list) do
                sec:AddScriptButton(item)
            end
        else
            sec:AddLabel(secInfo.EmptyText)
        end
    end

    -- Tự động hỗ trợ bất kỳ mục (Section) mới nào được thêm vào Database sau này
    for secKey, list in pairs(gameData) do
        if not handledKeys[secKey] and type(list) == "table" then
            local extraSec = createCollapsibleSection(tab, tostring(secKey), false)
            if #list > 0 then
                for _, item in ipairs(list) do
                    extraSec:AddScriptButton(item)
                end
            else
                extraSec:AddLabel("Hiện chưa có script cho mục này.")
            end
        end
    end

    return tab
end

-- 1. Anti-AFK (Chống Kick Khi Treo Máy Quá 20 Phút)
local antiAfkEnabled = false
local antiAfkConn = nil

local function setAntiAfk(state)
    antiAfkEnabled = state
    if antiAfkEnabled then
        local vu = nil

            end)
        end
        if not antiAfkThread then
            antiAfkThread = task.spawn(function()
                while antiAfkEnabled do
                    task.wait(600)
                    if antiAfkEnabled and vu then
                        pcall(function()
                            vu:CaptureController()
                            vu:ClickButton2(Vector2.new(0, 0))
                        end)

            end)
        end

    else

            antiAfkConn:Disconnect()

        end
        antiAfkThread = nil
        notify("Anti-AFK", "Anti-AFK: ĐÃ TẮT.", 3)
    end
end

-- 2. FPS Boost (Tối Ưu Hóa Đồ Họa & Giảm Lag)
local fpsBoostEnabled = false
local fpsDescendantConn = nil

local function optimizeInstance(inst)
    pcall(function()
        if inst:IsA("BasePart") then
            inst.Material = Enum.Material.SmoothPlastic
            inst.Reflectance = 0
            inst.CastShadow = false
        elseif inst:IsA("Decal") or inst:IsA("Texture") then
            inst.Transparency = 0.5
        elseif inst:IsA("ParticleEmitter") or inst:IsA("Trail") then
            inst.Lifetime = NumberRange.new(0)
            inst.Enabled = false
        elseif inst:IsA("Fire") or inst:IsA("Smoke") or inst:IsA("Sparkles") then
            inst.Enabled = false
        elseif inst:IsA("Explosion") then
            inst.BlastPressure = 1
            inst.BlastRadius = 1
        elseif inst:IsA("Highlight") then
            inst.Enabled = false
        end
    end)
end

local function applyFPSBoost()
    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    end)

    pcall(function()
        local lighting = game:GetService("Lighting")
        lighting.GlobalShadows = false
        lighting.FogEnd = 9e9
        lighting.Brightness = 1
        lighting.ShadowSoftness = 0
        for _, v in ipairs(lighting:GetChildren()) do
            if v:IsA("PostProcessEffect")
            or v:IsA("BloomEffect")
            or v:IsA("BlurEffect")
            or v:IsA("DepthOfFieldEffect")
            or v:IsA("SunRaysEffect")
            or v:IsA("ColorCorrectionEffect") then
                v.Enabled = false
            end
        end
    end)

    pcall(function()
        local terrain = workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            terrain.WaterWaveSize = 0
            terrain.WaterWaveSpeed = 0
            terrain.WaterReflectance = 0
            terrain.WaterTransparency = 0
            if sethiddenproperty then
                sethiddenproperty(terrain, "Decoration", false)
            end
        end
    end)

    for _, v in ipairs(workspace:GetDescendants()) do
        optimizeInstance(v)
    end
end

local function setFPSBoost(state)
    fpsBoostEnabled = state
    if fpsBoostEnabled then
        applyFPSBoost()
        if not fpsDescendantConn then
            fpsDescendantConn = workspace.DescendantAdded:Connect(function(child)
                if fpsBoostEnabled then
                    optimizeInstance(child)
                end
            end)
        end
        notify("FPS Boost", "Đã bật FPS Boost! Đồ họa đã được tối ưu mượt mà.", 3)
    else
        if fpsDescendantConn then
            fpsDescendantConn:Disconnect()
            fpsDescendantConn = nil
        end
        notify("FPS Boost", "Đã tắt theo dõi tối ưu FPS mới.", 3)
    end
end

-- 3. Server Hop (Đổi Server Ngẫu Nhiên)
local function serverHop()
    notify("Server Hop", "Đang tìm server khác...", 2.5)
    task.spawn(function()
        local placeId = game.PlaceId
        local success, servers = pcall(function()
            local url = "https://games.roblox.com/v1/games/" .. tostring(placeId) .. "/servers/Public?sortOrder=Asc&limit=100"
            return game:HttpGet(url)
        end)
        if success and servers then
            local decodeOk, parsed = pcall(function()
                return HttpService:JSONDecode(servers)
            end)
            if decodeOk and parsed and parsed.data then
                for _, s in ipairs(parsed.data) do
                    if s.id ~= game.JobId and (tonumber(s.playing) or 0) < (tonumber(s.maxPlayers) or 0) then
                        TeleportService:TeleportToPlaceInstance(placeId, s.id, LocalPlayer)
                        return
                    end
                end
            end
        end
        TeleportService:Teleport(placeId, LocalPlayer)
    end)
end

local function fixIcons()
    for _, t in pairs(rawWin._tabs) do
        if t._icon and t._isCustomIcon then
            t._icon.ImageColor3 = Color3.fromRGB(255, 255, 255)
            t._icon.ImageTransparency = 0
            if not t._icon:FindFirstChildWhichIsA("UICorner") then
                local c = Instance.new("UICorner")
                c.CornerRadius = UDim.new(0, 4)
                c.Parent = t._icon
            end
        end
    end
end
